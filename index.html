using UnityEngine; public class VRHandDrawer : MonoBehaviour { public Transform indexTip; // 검지 손가락 끝 오브젝트 public GameObject linePrefab; private LineRenderer currentLine; void Update() { // 손가락이 인식되고 그리기 상태일 때 if (IsPinching()) { if (currentLine == null) { GameObject lineObj = Instantiate(linePrefab); currentLine = lineObj.GetComponent(); } AddPoint(indexTip.position); } else { currentLine = null; } } void AddPoint(Vector3 position) { currentLine.positionCount++; currentLine.SetPosition(currentLine.positionCount - 1, position); } bool IsPinching() { // 엄지와 검지 끝 사이의 거리가 일정 값 이하인지 확인 return false; } }
